19def read_document(filename: str, data: bytes) -> str: 

20 name = (filename or "").lower() 

21 if name.endswith(".pdf"): 

22 from pypdf import PdfReader 

23 reader = PdfReader(io.BytesIO(data)) 

24 txt = "\n".join((p.extract_text() or "") for p in reader.pages[:40]) 

25 elif name.endswith(".docx"): 

26 import docx 

27 d = docx.Document(io.BytesIO(data)) 

28 parts = [p.text for p in d.paragraphs] 

29 for t in d.tables: 

30 for row in t.rows: 

31 parts.append(" | ".join(c.text for c in row.cells)) 

32 txt = "\n".join(parts) 

33 elif name.endswith((".txt", ".md")): 

34 txt = data.decode("utf-8", errors="replace") 

35 else: 

36 raise ValueError("Formato não suportado. Use PDF, DOCX, TXT ou MD.") 

37 return txt.strip()[:MAX_CHARS]

41def _sentences(text: str) -> list[str]: 

42 """Divide em orações (frases e trechos entre vírgulas) para decidir obrigatória/desejável e nível.""" 

43 return [s for s in re.split(r"(?<=[.;:!?])\s+|\n+|,\s*|\s+mas\s+", text) if s.strip()]

46def _short_forms(s: SGD) -> dict[str, str]: 

47 """Nomes curtos (<=2 letras) só casam com grafia exata para evitar falso positivo.""" 

48 return {"JS": "JavaScript", "TS": "TypeScript", "UX": "UX/UI", "UI": "UX/UI", "Go": "Go", "Golang": "Go", "C#": "C#", "c#": "C#", "C++": "C++", "c++": "C++"}

51def rules_extract(s: SGD, text: str) -> dict: 

52 ftext = fold(text) 

53 found: dict[str, dict] = {} 

54 keys = sorted((k for k in s.name_to_ids if len(k) > 2), key=len, reverse=True) 

55 sents = _sentences(text) 

56 

57 def add(canon: str, pos_text: str): 

58 if canon in found: 

59 return 

60 # se a skill aparece em várias orações, usa a que fala de requisito (obrigatória, desejável, nível…) 

61 cands = [fold(x) for x in sents if fold(pos_text) in fold(x)] 

62 sent = next((x for x in cands if re.search(CUES, x)), cands[0] if cands else "") 

63 desej = bool(re.search(r"desejav|diferencial|\bplus\b|nice to have|seria bom|opcional", sent)) 

64 if re.search(r"avancad|dominio|experiencia solida|solidos? conhecimentos?|profund|expert", sent): 

65 nivel = "avancado" 

66 elif re.search(r"basic|nocoes|nocao|introdut", sent): 

67 nivel = "basico" 

68 else: 

69 nivel = "basico" if desej else "intermediario" 

70 found[canon] = {"nome": canon, "obrigatoria": not desej, "nivel": nivel} 

71 

72 spans: list[tuple[int, int]] = [] 

73 for k in keys: # mais longos primeiro; ignora ocorrências dentro de um nome maior já casado 

74 for mt in re.finditer(r"(?<![a-z0-9#+])" + re.escape(k) + r"(?![a-z0-9#+])", ftext): 

75 a, b = mt.span() 

76 if any(x <= a and b <= y for x, y in spans): 

77 continue 

78 spans.append((a, b)) 

79 canon = s.canonical(k) 

80 if canon: 

81 add(canon, text[a:b]) 

82 for short, canon in _short_forms(s).items(): 

83 if re.search(r"(?<![\w#+])" + re.escape(short) + r"(?![\w#+])", text) and s.canonical(canon): 

84 add(s.canonical(canon), short) 

85 

86 out = {"skills": list(found.values())} 

87 m = re.search(r"(\d{1,2})\s*(meses|mes)\b", ftext) 

88 w = re.search(r"(\d{1,2})\s*semanas", ftext) 

89 if m: 

90 out["duracao_meses"] = int(m.group(1)) 

91 elif w: 

92 out["duracao_meses"] = max(1, math.ceil(int(w.group(1)) / 4)) 

93 p = re.search(r"(\d{1,3})\s*%", ftext) 

94 if p and 0 < int(p.group(1)) <= 100: 

95 out["disponibilidade_min"] = int(p.group(1)) 

96 elif re.search(r"tempo integral|dedicacao (integral|exclusiva|total)|full.?time|100 ?por ?cento", ftext): 

97 out["disponibilidade_min"] = 100 

98 elif re.search(r"meio periodo|meio-periodo|part.?time|parcial", ftext): 

99 out["disponibilidade_min"] = 50 

100 sens = [o for o in s.options["senioridades"] if re.search(r"\b" + re.escape(fold(o)), ftext)] 

101 if sens: 

102 out["senioridades"] = sens 

103 for campo, opts in (("tipo_servico", s.options["tipos_servico"]), ("metodologia", s.options["metodologias"])): 

104 hit = next((o for o in opts if re.search(r"\b" + re.escape(fold(o)), ftext)), None) 

105 if hit: 

106 out[campo] = hit 

107 ini = re.search(r"\b(" + "|".join(MESES_PT) + r")[a-z]*\.?\s*(?:de\s*|/)?\s*(20\d\d)", ftext) or \ 

108 re.search(r"\b(0?[1-9]|1[0-2])/(20\d\d)\b", ftext) 

109 if ini: 

110 mes = ini.group(1) 

111 mes = MESES_PT.index(mes[:3]) + 1 if not mes.isdigit() else int(mes) 

112 out["inicio"] = f"{ini.group(2)}-{mes:02d}" 

113 out["texto"] = text.strip()[:1500] 

114 return out

123def ai_extract(s: SGD, text: str, prazo: float = 25) -> dict: 

124 schema = { 

125 "skills": [{"nome": "nome do catálogo", "obrigatoria": True, "nivel": "basico|intermediario|avancado"}], 

126 "duracao_meses": "int ou null", "disponibilidade_min": "int 0-100 ou null", "inicio": "AAAA-MM ou null", 

127 "senioridades": ["valores de SENIORIDADES"], "tipo_servico": "um de TIPOS_SERVICO ou null", 

128 "metodologia": "uma de METODOLOGIAS ou null", "resumo": "resumo do projeto em até 400 caracteres", 

129 } 

130 user = (f"CATÁLOGO: {json.dumps(s.catalog, ensure_ascii=False)}\n" 

131 f"SENIORIDADES: {s.options['senioridades']}\nTIPOS_SERVICO: {s.options['tipos_servico']}\n" 

132 f"METODOLOGIAS: {s.options['metodologias']}\n\nFORMATO: {json.dumps(schema, ensure_ascii=False)}\n\n" 

133 f"BRIEFING:\n\"\"\"\n{text}\n\"\"\"") 

134 out = llm.as_json(SYS_EXTRACT, user, prazo=prazo) 

135 out["texto"] = (out.pop("resumo", None) or text)[:1500] 

136 return out

139def extract(s: SGD, text: str, use_ai: bool = True) -> dict: 

140 origem, erro = "regras", None 

141 if use_ai and llm.available(): 

142 try: 

143 out, origem = ai_extract(s, text), "ia" 

144 except Exception as e: # noqa: BLE001 

145 log.warning("extração por IA falhou: %s", llm._limpa(str(e))[:300]) 

146 erro = f"IA indisponível ({llm._limpa(str(e))[:160]}); usadas regras." 

147 out = rules_extract(s, text) 

148 else: 

149 out = rules_extract(s, text) 

150 # normaliza contra o catálogo 

151 skills, fora = [], [] 

152 for sk in out.get("skills") or []: 

153 canon = s.canonical(sk.get("nome", "")) 

154 if canon: 

155 skills.append({"nome": canon, "obrigatoria": bool(sk.get("obrigatoria", True)), 

156 "nivel": fold(sk.get("nivel") or "intermediario")}) 

157 elif sk.get("nome"): 

158 fora.append(sk["nome"]) 

159 out["skills"] = skills 

160 out["skills_fora_catalogo"] = fora 

161 out["senioridades"] = [x for x in (out.get("senioridades") or []) if x in s.options["senioridades"]] 

162 for campo, opts in (("tipo_servico", "tipos_servico"), ("metodologia", "metodologias")): 

163 if out.get(campo) not in s.options[opts]: 

164 out[campo] = None 

165 out["origem"] = origem 

166 if erro: 

167 out["aviso"] = erro 

168 return out
