1"""Camada de dados do SGD — SOMENTE LEITURA. 

2 

3Fonte preferencial: Postgres do hackathon (sessão com default_transaction_read_only=on). 

4Fallback: snapshot SQLite aberto em modo ro/immutable. 

5Apenas SELECTs explícitos. Nenhum DDL/DML é emitido por este módulo. 

6"""

27BOOL_COLS = {"status", "deletado", "compartilha_profissionais", "pertence_pmo"}

30def _norm(rows, cols): 

31 out = [] 

32 for r in rows: 

33 d = dict(zip(cols, r)) 

34 for c in BOOL_COLS & d.keys(): 

35 d[c] = bool(d[c]) if d[c] is not None else None 

36 for c in ("data_admissao", "data_inicio", "data_termino", "dt_inicio_demanda", 

37 "dt_termino_demanda_prevista", "criado_em"): 

38 if c in d and d[c] is not None: 

39 d[c] = str(d[c])[:10] 

40 out.append(d) 

41 return out

44def _load_postgres() -> dict: 

45 import psycopg # import tardio: só exige o driver se usar Postgres 

46 with psycopg.connect(config.DATABASE_URL, connect_timeout=3, 

47 options="-c default_transaction_read_only=on") as con: 

48 con.read_only = True 

49 data = {} 

50 with con.cursor() as cur: 

51 for t, cols in TABLES.items(): 

52 cur.execute(f'SELECT {cols} FROM "InHire".{t}') 

53 data[t] = _norm(cur.fetchall(), [c.strip() for c in cols.split(",")]) 

54 return data

57def _load_sqlite() -> dict: 

58 path = config.SQLITE_PATH.resolve().as_posix() 

59 con = sqlite3.connect(f"file:{path}?mode=ro&immutable=1", uri=True) 

60 try: 

61 data = {} 

62 for t, cols in TABLES.items(): 

63 data[t] = _norm(con.execute(f"SELECT {cols} FROM {t}").fetchall(), 

64 [c.strip() for c in cols.split(",")]) 

65 return data 

66 finally: 

67 con.close()

70def load_all() -> tuple[dict, str]: 

71 """Carrega as tabelas do SGD em memória (base pequena: ~20 mil linhas).""" 

72 src = config.DATA_SOURCE 

73 if src in ("auto", "postgres") and config.DATABASE_URL and "SEU_TOKEN" not in config.DATABASE_URL: 

74 try: 

75 data = _load_postgres() 

76 log.info("SGD carregado do Postgres (somente leitura)") 

77 return data, "postgres" 

78 except Exception as e: # noqa: BLE001 

79 if src == "postgres": 

80 raise 

81 log.warning("Postgres indisponível (%s); usando snapshot SQLite", type(e).__name__) 

82 data = _load_sqlite() 

83 log.info("SGD carregado do snapshot SQLite (somente leitura)") 

84 return data, "sqlite"
